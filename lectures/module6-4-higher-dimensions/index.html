function vander(x, k=nothing)
    if isnothing(k)
        k = length(x)
    end
    m = length(x)
    V = ones(m, k)
    for j in 2:k
        V[:, j] = V[:, j-1] .* x
    end
    V
end

function vander_chebyshev(x, n=nothing)
    if isnothing(n)
        n = length(x) # Square by default
    end
    m = length(x)
    T = ones(m, n)
    if n > 1
        T[:, 2] = x
    end
    for k in 3:n
        T[:, k] = 2 * x .* T[:,k-1] - T[:, k-2]
    end
    T
end

function interp_nearest(x, s)
    A = zeros(length(s), length(x))
    for (i, t) in enumerate(s)
        loc = nothing
        dist = Inf
        for (j, u) in enumerate(x)
            if abs(t - u) < dist
                loc = j
                dist = abs(t - u)
            end
        end
        A[i, loc] = 1
    end
    A
end

function interp_error(ieval, x, xx, test)
    """Compute norm of interpolation error for function test
    using method interp_and_eval from points x to points xx.
    """
    A = ieval(x, xx)
    y = test.(x)
    yy = test.(xx)
    norm(A * y - yy, Inf)
end

function plot_convergence(ievals, ptspaces; xscale=:log10, yscale=:log10, maxpts=40)
    """Plot convergence rates for an interpolation scheme applied
    to a set of tests.
    """
    xx = LinRange(-1, 1, 100)
    ns = 2:maxpts
    fig = plot(title="Convergence",
        xlabel="Number of points",
        ylabel="Interpolation error",
        xscale=xscale,
        yscale=yscale,
        legend=:bottomleft,

                             for n in ns]

                    continue
                end
            end
        end
    end
    for k in [1, 2, 3]

end

How can we generalize all this?

We saw that to create a Vandermonde matrix, we choose a family of functions ϕj(x)\phi_j(x)ϕj​(x) and a set of points xix_ixi​, then create the matrix

Vij=ϕj(xi).V_{ij} = \phi_j(x_i) .Vij​=ϕj​(xi​).
(1)

What if we wanted to approximate integrals of functions?¶

Bij=∫(xi−1+xi)/2(xi+xi+1)/2ϕj(s)dsB_{ij} = \int_{(x_{i-1} + x_i)/2}^{(x_i + x_{i+1})/2} \phi_j(s) dsBij​=∫(xi−1​+xi​)/2(xi​+xi+1​)/2​ϕj​(s)ds

This leads to conservative reconstruction, which is an important part of finite volume methods, which are industry standard for shock dynamics.

What if we wanted to approximate derivatives of functions?¶

What if we instead computed derivatives?

Aij=ϕj′(xi)A_{ij} = \phi_j'(x_i)Aij​=ϕj′​(xi​)
(3)

If we have the canonical, monomial basis, say up to third degree:

[1∣ x∣ x2 ∣ x3]\Bigg[ 1 \Bigg|\, x \Bigg|\, x^2 \,\Bigg|\, x^3 \Bigg][1∣∣​x∣∣​x2∣∣​x3]

Then the derivative looks like:

[0∣ 1∣ 2x ∣ 3x2]\Bigg[ 0 \Bigg|\, 1 \Bigg|\, 2x \,\Bigg|\, 3x^2 \Bigg][0∣∣​1∣∣​2x∣∣​3x2]

function diff_monomial(x)
    n = length(x)
    A = zeros(n, n)
    A[:,2] = one.(x)
    for j in 3:n
        A[:,j] = A[:,j-1] .* x * (j - 1) / (j - 2)
    end
    A
end

A stable basis: Chebyshev polynomials¶

Recall that Chebyshev polynomials satisfy a convenient recurrence:

T0(x)=1T1(x)=xTn+1(x)=2xTn(x)−Tn−1(x)\begin{split} T_0(x) &= 1 \\ T_1(x) &= x \\ T_{n+1}(x) &= 2 x T_n(x) - T_{n-1}(x) \end{split}T0​(x)T1​(x)Tn+1​(x)​=1=x=2xTn​(x)−Tn−1​(x)​
(6)

Derivatives of Chebyshev polynomials also satisfy a recurrence. So we can use this to write differentiation matrices: DDD to approximate first-derivatives and D2D^2D2 to approximate second-derivatives.

function chebdiff(x, n=nothing)
    T = vander_chebyshev(x, n)
    m, n = size(T)
    dT = zero(T)
    dT[:,2:3] = [one.(x) 4*x]
    for j in 3:n-1
        dT[:,j+1] = j * (2 * T[:,j] + dT[:,j-1] / (j-2))
    end
    ddT = zero(T)
    ddT[:,3] .= 4
    for j in 3:n-1
        ddT[:,j+1] = j * (2 * dT[:,j] + ddT[:,j-1] / (j-2))
    end
    T, dT, ddT
end

Solving a BVP with Chebyshev collocation¶

In our Solving Systems and PDEs lecture, we already saw a boundary value problem (BVP). We asked ourselves to find a function u(x)u(x)u(x) satisfying an equation like:

−uxx(x)=f(x)-u_{xx}(x) = f(x)−uxx​(x)=f(x)
(7)

subject to Boundary Conditions. This time, let’s use the following boundary conditions:

u(−1)=a and u′(1)=bu(-1) = a \quad \textrm{ and } \quad u'(1) = bu(−1)=a and u′(1)=b
(8)

The above Boundary Conditions are mixed: at the left endpoint of the interval, we have the Dirichelet Boundary Condition (which imposes a condition on the function uuu itself). At the right endpoint of the interval, we have a Neumann Boundary Condition (which imposes a condition on the first derivative of the function).

We’ll use the “method of manufactured solutions”:

Choose u(x)=tanh⁡(2x)u(x) = \tanh(2x)u(x)=tanh(2x) and solve with the corresponding f(x)f(x)f(x). In practice, f(x)f(x)f(x) comes from the physics of the problem or observations, and you need to solve for the unknown u(x)u(x)u(x).

Suppose we want to approximate a function—or solution to a differential equation—uuu, using a polynomial u(x)≈pn(x)u(x) \approx p_n(x)u(x)≈pn​(x). In this case, we are going to use Chebyshev polynomials, so we can stick to the notation u(x)≈Tn(x)u(x) \approx T_n(x)u(x)≈Tn​(x).

Instead of choosing arbitrary points, Chebyshev collocation uses Chebyshev points. We have already seen that these points are clustered near -1 and 1, preventing oscillations associated with polynomial interpolation at uniformly spaced points (runge’s phenomenon).

Therefore −uxx(x)=f(x)-u_{xx}(x) = f(x)−uxx​(x)=f(x) becomes:

−D2u=f- D^2 \mathbf{u} = \mathbf{f}−D2u=f
(9)

where D2D^2D2 is the second-derivative matrix given by differentiating in this case the Chebyshev polynomials twice, using the recurrence relations for the derivatives of the Chebyshev polynomials.

function poisson_cheb(n, rhsfunc, leftbc=(0, zero), rightbc=(0, zero))
    x = CosRange(-1, 1, n)
    T, dT, ddT = chebdiff(x)
    L = -ddT
    rhs = rhsfunc.(x)
    for (index, deriv, func) in
            [(1, leftbc...), (n, rightbc...)]
        L[index,:] = (T, dT)[deriv+1][index,:]
        rhs[index] = func(x[index])
    end
    x, L / T, rhs
end

Fourier series and tensor product structure¶

For periodic data on the interval [−π,π)[-\pi, \pi)[−π,π), we can use a basis {1,sin⁡x,cos⁡x,sin⁡2x,cos⁡2x,… }\{ 1, \sin x, \cos x, \sin 2x, \cos 2x, \dotsc\}{1,sinx,cosx,sin2x,cos2x,…}, which is equivalent to {1,eix,ei2x,… }\{ 1, e^{ix}, e^{i2x}, \dotsc \}{1,eix,ei2x,…} with suitable complex coefficients.

If we’re given equally spaced points on the interval, the Vandermonde matrix VVV (with suitable scaling) is unitary (like orthogonal for complex matrices) and can be applied in O(nlog⁡n)O(n \log n)O(nlogn) (with small constants) using the Fast Fourier Transform. This also works for Chebyshev polynomials sampled on CosRange points.

function chebyshev_regress_eval(x, xx, n)
    V = vander_chebyshev(x, n)
    @show cond(V)
    vander_chebyshev(xx, n) / V
end
