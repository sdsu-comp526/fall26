using LinearAlgebra
using Plots
default(linewidth=4, legendfontsize=12)

function vander(x, k=nothing)
    if isnothing(k)
        k = length(x)
    end
    m = length(x)
    V = ones(m, k)
    for j in 2:k
        V[:, j] = V[:, j-1] .* x
    end
    V
end

function gram_schmidt_classical(A)
    m, n = size(A)
    Q = zeros(m, n)
    R = zeros(n, n)
    for j in 1:n
        v = A[:,j]
        R[1:j-1,j] = Q[:,1:j-1]' * v
        v -= Q[:,1:j-1] * R[1:j-1,j]
        R[j,j] = norm(v)
        Q[:,j] = v / R[j,j]
    end
    Q, R
end

function qr_householder(A)
    m, n = size(A)
    R = copy(A)
    V = [] # list of reflectors
    for j in 1:n
        v = copy(R[j:end, j])
        v[1] += sign(v[1]) * norm(v) # <--- here we pick the sign of v so that moves it the largest distance
        v = normalize(v)
        R[j:end,j:end] -= 2 * v * v' * R[j:end,j:end]
        push!(V, v)
    end
    V, R
end

function qr_chol(A)
    R = cholesky(A' * A).U
    Q = A / R
    Q, R
end

function qr_chol2(A)
    Q, R = qr_chol(A)
    Q, R1 = qr_chol(Q)
    Q, R1 * R
end

function peanut()
    theta = LinRange(0, 2*pi, 50)
    r = 1 .+ .4*sin.(3*theta) + .6*sin.(2*theta)
    r' .* [cos.(theta) sin.(theta)]'
end

function circle()
    theta = LinRange(0, 2*pi, 50)
    [cos.(theta) sin.(theta)]'
end

function Aplot(A)
    "Plot a transformation from X to Y"
    X = peanut()
    Y = A * X
    p = scatter(X[1,:], X[2,:], label="in")
    scatter!(p, Y[1,:], Y[2,:], label="out")
    X = circle()
    Y = A * X
    q = scatter(X[1,:], X[2,:], label="in")
    scatter!(q, Y[1,:], Y[2,:], label="out")
    plot(p, q, layout=2, aspect_ratio=:equal)
end

Recap on Condition number via SVD¶

Recall the SVD decomposition

U[σmax⁡⋱σmin⁡]⏞ΣVT=AU \overbrace{\begin{bmatrix} \sigma_{\max} && \\ & \ddots & \\ && \sigma_{\min} \end{bmatrix}}^{\Sigma} V^T = AU⎣⎡​σmax​​⋱​σmin​​⎦⎤​​Σ​VT=A
(1)

Previously, we saw the matrix norm induced by vector norms.

Here, we consider a special case: the matrix norm induced by the vector ppp-norm.

For the particular case of p=2p=2p=2, the vector norm is the simple Euclidean norm or l2l_2l2​-norm for vectors, and the matrix norm is called the spectral norm. The spectral norm of a matrix AAA is the largest singular value of AAA:

∥A∥2=σmax⁡≡λmax⁡(A⋆A)\begin{align} \lVert A \rVert_2 &= \sigma_{\max} \equiv \sqrt{\lambda_{\max}(A^{\star}A)} \end{align}∥A∥2​​=σmax​≡λmax​(A⋆A)​​​
(2)

And, we have seen that:

κ(A)=σmax⁡σmin⁡=cond(A)\kappa(A) = \frac{\sigma_{\max}}{\sigma_{\min}} = \texttt{cond}(A)κ(A)=σmin​σmax​​=cond(A)

Forces between stars¶

Consider Newton’s law of universal gravitation between two bodies with spatial extent (i.e., not point masses), then we can write the gravitational force from a star at position x2x_2x2​ acting on a star at position x1x_1x1​,

F1,2=Gm1m2∥x2−x1∥3(x2−x1)F_{1,2} = G \frac{m_1 m_2}{\lVert \mathbf{x}_2 - \mathbf{x}_1 \rVert^3} (\mathbf{x}_2 - \mathbf{x}_1)F1,2​=G∥x2​−x1​∥3m1​m2​​(x2​−x1​)
(4)

where m1m_1m1​ and m2m_2m2​ are the masses of each star, respectively.

function gravitational_force(g1, g2)
    m = size(g1, 1)
    n = size(g2, 1)
    F = zeros(3*m, n)
    for i in 0:m-1
        for j in 1:n
            r = g2[j,:] - g1[1+i,:]
            F[1+3*i:3*(i+1),j] = r / norm(r)^3
        end
    end
    F
end
